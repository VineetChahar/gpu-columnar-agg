cmake_minimum_required(VERSION 3.18)
project(gpu_columnar_agg LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 75)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(CUDAToolkit REQUIRED)

# warnings are errors everywhere; CPU code is built -O3 -march=native (fair CPU baseline)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native")
set(CMAKE_CUDA_FLAGS_RELEASE "-O3")
set(WARN_CXX -Wall -Wextra -Werror)
set(WARN_CUDA -Werror=all-warnings -lineinfo --expt-relaxed-constexpr -Xcompiler=-Wall,-Wextra,-Werror,-O3,-march=native)

add_library(cpu STATIC src/datagen.cpp src/cpu/reference.cpp src/cpu/engine.cpp src/cpu/join_cpu.cpp)
target_compile_options(cpu PRIVATE ${WARN_CXX})
find_package(Threads REQUIRED)
target_link_libraries(cpu PUBLIC Threads::Threads)

file(GLOB KERNELS src/kernels/*.cu)
add_executable(bench bench/bench.cu bench/micro.cu bench/stream.cu bench/join.cu bench/tpch.cu src/gpu/dev.cu ${KERNELS})
target_compile_options(bench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${WARN_CUDA}>)
find_library(NVML_LIB nvidia-ml HINTS /usr/local/nvidia/lib64 ${CUDAToolkit_LIBRARY_DIR}/stubs ${CUDAToolkit_LIBRARY_ROOT}/lib64/stubs /usr/lib/x86_64-linux-gnu /usr/local/cuda/lib64/stubs REQUIRED)
target_link_libraries(bench PRIVATE cpu CUDA::cudart ${NVML_LIB})
